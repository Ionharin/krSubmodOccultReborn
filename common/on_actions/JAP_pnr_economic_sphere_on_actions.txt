on_actions = {
	# Keeps the Yen Sphere in sync with Japan's subjects and faction members, and refreshes its bonuses.
	# Weekly and Japan-only instead of a global on_daily: new members join within a week at a fraction of the cost.
	on_weekly_JAP = {
		effect = {
			if = {
				limit = {
					is_economic_sphere_leader = yes
					has_dynamic_modifier = { modifier = JAP_pnr_yen_economic_sphere_leader }
				}
				JAP_pnr_refresh_yen_economic_sphere_members = yes
				JAP_pnr_recalculate_yen_economic_sphere_economic_integration = yes
			}
		}
	}
}
