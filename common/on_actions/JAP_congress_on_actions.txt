on_actions = {
	on_startup = {
		effect = {
			JAP = {
				if = {
					limit = { exists = yes }
					JAP_congress_initialize_values = yes
					JAP_pnr_election_ensure_projected_results = yes
					JAP_pnr_initialize_interservice_budget = yes
					if = {
						limit = {
							has_completed_focus = JAP_pnr_68th_sessions_imperial_diet
							NOT = { has_country_flag = JAP_pnr_election_completed }
						}
						set_country_flag = pnr_congress_temporarily_unable
						set_country_flag = JAP_pnr_election_in_progress
					}
					if = {
						limit = { NOT = { has_country_flag = JAP_congress_annual_agenda_loop_started } }
						set_country_flag = JAP_congress_annual_agenda_loop_started
						country_event = { id = JAP_pnr_event.2 days = 365 }
					}
				}
				random_list = {
                    50 = { unlock_national_focus = JAP_pnr_join_forces_with_the_wakuhara_faction }
    				50 = { unlock_national_focus = JAP_pnr_leaning_toward_suzuki_faction }
                }
			}
		}
	}

	on_daily = {
		effect = {
			JAP = {
				if = {
					limit = {
						exists = yes
						JAP_congress_interface_is_open = yes
					}
					JAP_congress_rebuild_seats = yes
				}
			}
		}
	}
}
