on_actions = {
	on_startup = { #一系列初始化，隐藏事件会循环初始化议会的进程
		effect = {
			JAP = {
				JAP_congress_initialize_values = yes

				JAP_pnr_initialize_operation_launched_modifier_variables = yes

				if = {
					limit = {
						has_completed_focus = JAP_pnr_68th_sessions_imperial_diet
						NOT = { has_country_flag = JAP_pnr_election_completed }
					}
					set_country_flag = pnr_congress_temporarily_unable
					set_country_flag = JAP_pnr_election_in_progress
					}
				if = {
					limit = { NOT = { has_country_flag = JAP_congress_annual_agenda_loop_started } }
					set_country_flag = JAP_congress_annual_agenda_loop_started
					country_event = { id = JAP_pnr_hidden.3 days = 365 }
				}
				random_list = {
                	50 = { unlock_national_focus = JAP_pnr_join_forces_with_the_wakuhara_faction }
    				50 = { unlock_national_focus = JAP_pnr_leaning_toward_suzuki_faction }
                }
			}
		}
	}

	on_monthly_JAP = { #给昭和大帝用的系统
		effect = {
			if = {
				limit = { exists = yes }

				set_variable = { JAP.JAP_pnr_oriential_company_modifier_industrial_factory_donations_var = THIS.num_subjects }
				force_update_dynamic_modifier = yes
			}
		}
	}

	on_daily_JAP = { #每天刷新席位
		effect = {
				if = {
					limit = {
						exists = yes
						JAP_congress_interface_is_open = yes
					}
					JAP_congress_rebuild_seats = yes
				}
		}
	}

	on_weekly_JAP = { #每周刷新经济圈
		effect = {
			if = {
				limit = {
					is_economic_sphere_leader = yes
					has_dynamic_modifier = { modifier = JAP_pnr_yen_economic_sphere_leader }
				}
				JAP_pnr_refresh_yen_economic_sphere_members = yes
				JAP_pnr_recalculate_yen_economic_sphere_economic_integration = yes
			}
		}
	}

	on_civil_war_end = { # 为了死了妈的泰国内战判断
		effect = { # For SIA civil War
			if = {
				limit = { original_tag = SIA }

					limit = {
						SIA = {
							has_country_flag = SIA_REBEL
							NOT = { has_global_flag = SIA_phibunsongkhram_is_royalist }
						}
					}
					JAP = { country_event = { id = JAP_pnr_military_events.22 days = 3 } }
				}
			}      
		}
	}

		effect = {
			if = {
				limit = {
					original_tag = JAP
					is_faction_leader = yes
					any_enemy_country = {
						is_chinese_tag = yes
						is_subject = no
					}
				}

			}
		}
	}
