on_actions = {
	on_daily = {
		effect = {
			if = { # Owen Lattimore coming!
				limit = {
					tag = QIE
					date > 1936.11.1
					date < 1936.11.30
					NOT = { has_country_flag = QIE_ws_leader_30_triggered }
				}
				set_country_flag = QIE_ws_leader_30_triggered
				country_event = { id = QIE_ws_leader.30 days = 0 }
			}
			if = { # In fact, Hungary is Shanxi,right?
				limit = {
					tag = QIE
					NOT = { has_country_flag = QIE_ws_normal_event_10_triggered }
					NOT = { country_exists = HUN }
					HUN = {
						any_core_state = { is_controlled_by = QIE }
						all_core_state = { is_controlled_by = QIE }
					}
				}
				set_country_flag = QIE_ws_normal_event_10_triggered
				country_event = { id = QIE_ws_normal_events.10 days = 0 }
			}
			if = { # industry building post-war industrial promotion
				limit = {
					tag = QIE
					has_idea = QIE_wushan_jiangzhou_industrial_promotion
				}
				add_to_variable = { QIE_wushan_jiangzhou_factory_timer = 1 }
				if = {
					limit = {
						OR = {
							AND = {
								has_country_flag = QIE_wushan_jiangzhou_industrial_promotion_upgraded
								check_variable = { QIE_wushan_jiangzhou_factory_timer > 79 }
							}
							AND = {
								NOT = { has_country_flag = QIE_wushan_jiangzhou_industrial_promotion_upgraded }
								check_variable = { QIE_wushan_jiangzhou_factory_timer > 61 }
							}
						}
						any_core_state = {
							is_controlled_by = PREV
							free_building_slots = { building = industrial_complex size > 0 include_locked = no }
							is_valid_state_for_development = yes
						}
					}
					set_variable = { QIE_wushan_jiangzhou_factory_timer = 0 }
					if = {
						limit = { has_country_flag = QIE_wushan_jiangzhou_industrial_promotion_upgraded }
						random_controlled_state = {
							limit = {
								is_core_of = PREV
								free_building_slots = { building = industrial_complex size > 0 include_locked = no }
								is_valid_state_for_development = yes
							}
							random_list = {
								50 = {
									add_building_construction = { type = industrial_complex level = 1 instant_build = yes }
								}
								50 = {
									add_building_construction = { type = arms_factory level = 1 instant_build = yes }
								}
							}
						}
						random_controlled_state = {
							limit = {
								is_core_of = PREV
								free_building_slots = { building = industrial_complex size > 0 include_locked = no }
								is_valid_state_for_development = yes
							}
							random_list = {
								50 = {
									add_building_construction = { type = industrial_complex level = 1 instant_build = yes }
								}
								50 = {
									add_building_construction = { type = arms_factory level = 1 instant_build = yes }
								}
							}
						}
					}
					else = {
						random_controlled_state = {
							limit = {
								is_core_of = PREV
								free_building_slots = { building = industrial_complex size > 0 include_locked = no }
								is_valid_state_for_development = yes
							}
							random_list = {
								50 = {
									add_building_construction = { type = industrial_complex level = 1 instant_build = yes }
								}
								50 = {
									add_building_construction = { type = arms_factory level = 1 instant_build = yes }
								}
							}
						}
					}
				}
			}
			if = { # electrification spreads to neighbouring core states
				limit = {
					tag = QIE
					has_country_flag = QIE_wushan_electrification_spread_enabled
				}
				add_to_variable = { QIE_wushan_electrification_spread_timer = 1 }
				if = {
					limit = {
						check_variable = { QIE_wushan_electrification_spread_timer > 89 }
						any_controlled_state = {
							has_dynamic_modifier = {
								modifier = QIE_wushan_electrification_modifier
							}
							any_neighbor_state = {
								is_controlled_by = ROOT
								is_core_of = ROOT
								NOT = {
									has_dynamic_modifier = {
										modifier = QIE_wushan_electrification_modifier
									}
								}
							}
						}
					}
					set_variable = { QIE_wushan_electrification_spread_timer = 0 }
					random_controlled_state = {
						limit = {
							has_dynamic_modifier = {
								modifier = QIE_wushan_electrification_modifier
							}
							any_neighbor_state = {
								is_controlled_by = ROOT
								is_core_of = ROOT
								NOT = {
									has_dynamic_modifier = {
										modifier = QIE_wushan_electrification_modifier
									}
								}
							}
						}
						random_neighbor_state = {
							limit = {
								is_controlled_by = ROOT
								is_core_of = ROOT
								NOT = {
									has_dynamic_modifier = {
										modifier = QIE_wushan_electrification_modifier
									}
								}
							}
							add_dynamic_modifier = {
								modifier = QIE_wushan_electrification_modifier
							}
						}
					}
				}
			}
			if = { # Analyze seances status
				limit = {
					tag = QIE
					has_completed_focus = QIE_seances_restart_in_beijing
					has_variable = god_circle_focus_var
					check_variable = { god_circle_tier < 3 }
					NOT = {
						has_country_flag = QIE_recently_changed_god_circle_focus
					}
				}
				add_to_variable = {
					god_circle_tier = 1
				}
				set_country_flag = {
					flag = QIE_recently_changed_god_circle_focus
					days = 70
					value = 1
				}
				QIE_update_god_circle_tier = yes
			}
		}
	}
}
