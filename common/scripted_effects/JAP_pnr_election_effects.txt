### JAP 1936 Election Effects ###

### Final Result Seat Tuning ###
# The projected seats (JAP_pnr_election_party_seats^n) only move during the campaign for display.
# The final result is decided by JAP_pnr_election_party_true_factor^n (see JAP_pnr_election_triggers.txt)
# and the seats below. Every distribution adds up to @JAP_PNR_ELECTION_TOTAL_SEATS.
@JAP_PNR_ELECTION_TOTAL_SEATS = 466
@JAP_PNR_ELECTION_WINNER_SEATS = 244
@JAP_PNR_ELECTION_SECOND_SEATS = 106
@JAP_PNR_ELECTION_THIRD_SEATS = 82
@JAP_PNR_ELECTION_REST_SEATS_1 = 14
@JAP_PNR_ELECTION_REST_SEATS_2 = 12
@JAP_PNR_ELECTION_REST_SEATS_3 = 6
@JAP_PNR_ELECTION_REST_SEATS_4 = 2
# Shakai Taishuto victory: the winner seats are split between its social democrat (4) and social liberal (5) wings
@JAP_PNR_ELECTION_SHAKAI_TAISHUTO_WIN_SEATS_4 = 135
@JAP_PNR_ELECTION_SHAKAI_TAISHUTO_WIN_SEATS_5 = 109
# Seimin Coalition victory
@JAP_PNR_ELECTION_COALITION_SEATS_7 = 147
@JAP_PNR_ELECTION_COALITION_SEATS_6 = 125
@JAP_PNR_ELECTION_COALITION_REST_SEATS_1 = 76
@JAP_PNR_ELECTION_COALITION_REST_SEATS_2 = 64
@JAP_PNR_ELECTION_COALITION_REST_SEATS_3 = 34
@JAP_PNR_ELECTION_COALITION_REST_SEATS_4 = 16
@JAP_PNR_ELECTION_COALITION_REST_SEATS_5 = 4
# Fringe parties never run for government; these scores keep them at the bottom of the rest-seat ranking
@JAP_PNR_ELECTION_SYNDICALIST_TRUE_FACTOR = -50
@JAP_PNR_ELECTION_INDEPENDENT_TRUE_FACTOR = -20
# Halving divisor for Shakai Taishuto seats when it does not win
@JAP_PNR_ELECTION_SHAKAI_TAISHUTO_WINGS = 2
# Starting score for the rest-seat ranking; lower than any real score
@JAP_PNR_ELECTION_RANK_FLOOR = -1000

	set_variable = { JAP_pnr_election_party_seats^2 = 7 }
	set_variable = { JAP_pnr_election_party_seats^3 = 0 }
	set_variable = { JAP_pnr_election_party_seats^4 = 31 }
	set_variable = { JAP_pnr_election_party_seats^5 = 30 }
	set_variable = { JAP_pnr_election_party_seats^6 = 135 }
	set_variable = { JAP_pnr_election_party_seats^7 = 174 }
	set_variable = { JAP_pnr_election_party_seats^8 = 45 }
	set_variable = { JAP_pnr_election_party_seats^9 = 12 }
	set_variable = { JAP_pnr_election_party_seats^10 = 32 }

	set_variable = { JAP_pnr_election_party_seats_initial^2 = 7 }
	set_variable = { JAP_pnr_election_party_seats_initial^4 = 31 }
	set_variable = { JAP_pnr_election_party_seats_initial^5 = 30 }
	set_variable = { JAP_pnr_election_party_seats_initial^6 = 135 }
	set_variable = { JAP_pnr_election_party_seats_initial^7 = 174 }
	set_variable = { JAP_pnr_election_party_seats_initial^8 = 45 }
	set_variable = { JAP_pnr_election_party_seats_initial^9 = 12 }
	set_variable = { JAP_pnr_election_party_seats_initial^10 = 32 }

	JAP_pnr_election_refresh_projected_changes = yes
}

JAP_pnr_election_ensure_projected_results = {
	if = {
		limit = { NOT = { has_country_flag = JAP_pnr_election_projected_initialized } }
		JAP_pnr_election_initialize_projected_results = yes
	}
}

JAP_pnr_election_refresh_projected_donor = {
	set_variable = { JAP_pnr_election_donor_party = 2 }
	set_variable = { JAP_pnr_election_donor_party_seats = JAP_pnr_election_party_seats^2 }

	if = { limit = { check_variable = { JAP_pnr_election_party_seats^4 > JAP_pnr_election_donor_party_seats } } set_variable = { JAP_pnr_election_donor_party = 4 } set_variable = { JAP_pnr_election_donor_party_seats = JAP_pnr_election_party_seats^4 } }
	if = { limit = { check_variable = { JAP_pnr_election_party_seats^5 > JAP_pnr_election_donor_party_seats } } set_variable = { JAP_pnr_election_donor_party = 5 } set_variable = { JAP_pnr_election_donor_party_seats = JAP_pnr_election_party_seats^5 } }
	if = { limit = { check_variable = { JAP_pnr_election_party_seats^6 > JAP_pnr_election_donor_party_seats } } set_variable = { JAP_pnr_election_donor_party = 6 } set_variable = { JAP_pnr_election_donor_party_seats = JAP_pnr_election_party_seats^6 } }
	if = { limit = { check_variable = { JAP_pnr_election_party_seats^7 > JAP_pnr_election_donor_party_seats } } set_variable = { JAP_pnr_election_donor_party = 7 } set_variable = { JAP_pnr_election_donor_party_seats = JAP_pnr_election_party_seats^7 } }
	if = { limit = { check_variable = { JAP_pnr_election_party_seats^8 > JAP_pnr_election_donor_party_seats } } set_variable = { JAP_pnr_election_donor_party = 8 } set_variable = { JAP_pnr_election_donor_party_seats = JAP_pnr_election_party_seats^8 } }
}

### Final Result Helpers ###

JAP_pnr_election_reset_final_seats = {
	set_variable = { JAP_pnr_election_final_seats^1 = 0 }
	set_variable = { JAP_pnr_election_final_seats^2 = 0 }
	set_variable = { JAP_pnr_election_final_seats^3 = 0 }
	set_variable = { JAP_pnr_election_final_seats^4 = 0 }
	set_variable = { JAP_pnr_election_final_seats^5 = 0 }
	set_variable = { JAP_pnr_election_final_seats^6 = 0 }
	set_variable = { JAP_pnr_election_final_seats^7 = 0 }
	set_variable = { JAP_pnr_election_final_seats^8 = 0 }
	set_variable = { JAP_pnr_election_final_seats^9 = 0 }
	set_variable = { JAP_pnr_election_final_seats^10 = 0 }
	JAP_pnr_election_clear_rest_pool = yes
}

JAP_pnr_election_clear_rest_pool = {
	clr_country_flag = JAP_pnr_election_rest_pool_2
	clr_country_flag = JAP_pnr_election_rest_pool_4
	clr_country_flag = JAP_pnr_election_rest_pool_8
	clr_country_flag = JAP_pnr_election_rest_pool_9
	clr_country_flag = JAP_pnr_election_rest_pool_10
}

# Seiyukai (7) and Minseito (6) take the second and third places by score; Seiyukai wins a tie.
JAP_pnr_election_assign_major_parties_by_score = {
	if = {
		limit = { check_variable = { JAP_pnr_election_party_true_factor^6 > JAP_pnr_election_party_true_factor^7 } }
		add_to_variable = { JAP_pnr_election_final_seats^6 = @JAP_PNR_ELECTION_SECOND_SEATS }
		add_to_variable = { JAP_pnr_election_final_seats^7 = @JAP_PNR_ELECTION_THIRD_SEATS }
	}
	else = {
		add_to_variable = { JAP_pnr_election_final_seats^7 = @JAP_PNR_ELECTION_SECOND_SEATS }
		add_to_variable = { JAP_pnr_election_final_seats^6 = @JAP_PNR_ELECTION_THIRD_SEATS }
	}
}

# Input: temp variable JAP_pnr_election_shakai_taishuto_share
# A non-winning Shakai Taishuto splits its seats evenly between parties 4 and 5; an odd seat goes to the independents (9).
JAP_pnr_election_add_shakai_taishuto_share = {
	set_temp_variable = { JAP_pnr_election_shakai_taishuto_odd_seat = JAP_pnr_election_shakai_taishuto_share }
	modulo_temp_variable = { JAP_pnr_election_shakai_taishuto_odd_seat = @JAP_PNR_ELECTION_SHAKAI_TAISHUTO_WINGS }
	set_temp_variable = { JAP_pnr_election_shakai_taishuto_half = JAP_pnr_election_shakai_taishuto_share }
	subtract_from_temp_variable = { JAP_pnr_election_shakai_taishuto_half = JAP_pnr_election_shakai_taishuto_odd_seat }
	divide_temp_variable = { JAP_pnr_election_shakai_taishuto_half = @JAP_PNR_ELECTION_SHAKAI_TAISHUTO_WINGS }
	add_to_variable = { JAP_pnr_election_final_seats^4 = JAP_pnr_election_shakai_taishuto_half }
	add_to_variable = { JAP_pnr_election_final_seats^5 = JAP_pnr_election_shakai_taishuto_half }
	add_to_variable = { JAP_pnr_election_final_seats^9 = JAP_pnr_election_shakai_taishuto_odd_seat }
}

# Input: temp variable JAP_pnr_election_slot_seats
# Gives one rest-seat slot to the best-scoring party still flagged JAP_pnr_election_rest_pool_<n>, then removes it from the pool.
# Candidates are checked in ascending index order and replace the current best on an equal score, so ties go to the higher index.
# Party 4 stands for the whole Shakai Taishuto.
JAP_pnr_election_give_rest_slot = {
	set_temp_variable = { JAP_pnr_election_best_party = 0 }
	set_temp_variable = { JAP_pnr_election_best_factor = @JAP_PNR_ELECTION_RANK_FLOOR }
	if = {
		limit = {
			has_country_flag = JAP_pnr_election_rest_pool_2
			NOT = { check_variable = { JAP_pnr_election_party_true_factor^2 < JAP_pnr_election_best_factor } }
		}
		set_temp_variable = { JAP_pnr_election_best_party = 2 }
		set_temp_variable = { JAP_pnr_election_best_factor = JAP_pnr_election_party_true_factor^2 }
	}
	if = {
		limit = {
			has_country_flag = JAP_pnr_election_rest_pool_4
			NOT = { check_variable = { JAP_pnr_election_party_true_factor^4 < JAP_pnr_election_best_factor } }
		}
		set_temp_variable = { JAP_pnr_election_best_party = 4 }
		set_temp_variable = { JAP_pnr_election_best_factor = JAP_pnr_election_party_true_factor^4 }
	}
	if = {
		limit = {
			has_country_flag = JAP_pnr_election_rest_pool_8
			NOT = { check_variable = { JAP_pnr_election_party_true_factor^8 < JAP_pnr_election_best_factor } }
		}
		set_temp_variable = { JAP_pnr_election_best_party = 8 }
		set_temp_variable = { JAP_pnr_election_best_factor = JAP_pnr_election_party_true_factor^8 }
	}
	if = {
		limit = {
			has_country_flag = JAP_pnr_election_rest_pool_9
			NOT = { check_variable = { JAP_pnr_election_party_true_factor^9 < JAP_pnr_election_best_factor } }
		}
		set_temp_variable = { JAP_pnr_election_best_party = 9 }
		set_temp_variable = { JAP_pnr_election_best_factor = JAP_pnr_election_party_true_factor^9 }
	}
	if = {
		limit = {
			has_country_flag = JAP_pnr_election_rest_pool_10
			NOT = { check_variable = { JAP_pnr_election_party_true_factor^10 < JAP_pnr_election_best_factor } }
		}
		set_temp_variable = { JAP_pnr_election_best_party = 10 }
		set_temp_variable = { JAP_pnr_election_best_factor = JAP_pnr_election_party_true_factor^10 }
	}

	if = {
		limit = { check_variable = { JAP_pnr_election_best_party = 2 } }
		add_to_variable = { JAP_pnr_election_final_seats^2 = JAP_pnr_election_slot_seats }
		clr_country_flag = JAP_pnr_election_rest_pool_2
	}
	else_if = {
		limit = { check_variable = { JAP_pnr_election_best_party = 4 } }
		set_temp_variable = { JAP_pnr_election_shakai_taishuto_share = JAP_pnr_election_slot_seats }
		JAP_pnr_election_add_shakai_taishuto_share = yes
		clr_country_flag = JAP_pnr_election_rest_pool_4
	}
	else_if = {
		limit = { check_variable = { JAP_pnr_election_best_party = 8 } }
		add_to_variable = { JAP_pnr_election_final_seats^8 = JAP_pnr_election_slot_seats }
		clr_country_flag = JAP_pnr_election_rest_pool_8
	}
	else_if = {
		limit = { check_variable = { JAP_pnr_election_best_party = 9 } }
		add_to_variable = { JAP_pnr_election_final_seats^9 = JAP_pnr_election_slot_seats }
		clr_country_flag = JAP_pnr_election_rest_pool_9
	}
	else_if = {
		limit = { check_variable = { JAP_pnr_election_best_party = 10 } }
		add_to_variable = { JAP_pnr_election_final_seats^10 = JAP_pnr_election_slot_seats }
		clr_country_flag = JAP_pnr_election_rest_pool_10
	}
	else = {
		# Should never happen: every slot list is exactly as long as its pool. Keep the seat total intact anyway.
		log = "[GetDateText]: JAP election: empty rest pool, slot seats go to independents"
		add_to_variable = { JAP_pnr_election_final_seats^9 = JAP_pnr_election_slot_seats }
	}
}

### Final Result ###
# Sets JAP_pnr_election_winning_party and JAP_pnr_election_final_seats^n.
# Winner ids: 5 = Shakai Taishuto, 6 = Minseito, 7 = Seiyukai, 8 = Kokumin Domei, 10 = National Populist, 11 = Seimin Coalition.
# 12 (transcendent cabinet) is a special ending and is never produced here.
JAP_pnr_election_calculate_winner = {
	JAP_pnr_election_ensure_projected_results = yes
	JAP_pnr_election_reset_final_seats = yes

	if = {
		limit = { JAP_pnr_election_seimin_coalition_leads = yes }
		set_variable = { JAP_pnr_election_winning_party = 11 }
		add_to_variable = { JAP_pnr_election_final_seats^7 = @JAP_PNR_ELECTION_COALITION_SEATS_7 }
		add_to_variable = { JAP_pnr_election_final_seats^6 = @JAP_PNR_ELECTION_COALITION_SEATS_6 }
		set_country_flag = JAP_pnr_election_rest_pool_2
		set_country_flag = JAP_pnr_election_rest_pool_4
		set_country_flag = JAP_pnr_election_rest_pool_8
		set_country_flag = JAP_pnr_election_rest_pool_9
		set_country_flag = JAP_pnr_election_rest_pool_10
		JAP_pnr_election_distribute_coalition_rest_seats = yes
	}
	else_if = {
		limit = { JAP_pnr_election_party_10_leads = yes }
		set_variable = { JAP_pnr_election_winning_party = 10 }
		add_to_variable = { JAP_pnr_election_final_seats^10 = @JAP_PNR_ELECTION_WINNER_SEATS }
		JAP_pnr_election_assign_major_parties_by_score = yes
		set_country_flag = JAP_pnr_election_rest_pool_2
		set_country_flag = JAP_pnr_election_rest_pool_4
		set_country_flag = JAP_pnr_election_rest_pool_8
		set_country_flag = JAP_pnr_election_rest_pool_9
		JAP_pnr_election_distribute_standard_rest_seats = yes
	}
	else_if = {
		limit = { JAP_pnr_election_party_8_leads = yes }
		set_variable = { JAP_pnr_election_winning_party = 8 }
		add_to_variable = { JAP_pnr_election_final_seats^8 = @JAP_PNR_ELECTION_WINNER_SEATS }
		JAP_pnr_election_assign_major_parties_by_score = yes
		set_country_flag = JAP_pnr_election_rest_pool_2
		set_country_flag = JAP_pnr_election_rest_pool_4
		set_country_flag = JAP_pnr_election_rest_pool_9
		set_country_flag = JAP_pnr_election_rest_pool_10
		JAP_pnr_election_distribute_standard_rest_seats = yes
	}
	else_if = {
		limit = { JAP_pnr_election_party_7_leads = yes }
		set_variable = { JAP_pnr_election_winning_party = 7 }
		add_to_variable = { JAP_pnr_election_final_seats^7 = @JAP_PNR_ELECTION_WINNER_SEATS }
		add_to_variable = { JAP_pnr_election_final_seats^6 = @JAP_PNR_ELECTION_SECOND_SEATS }
		set_temp_variable = { JAP_pnr_election_shakai_taishuto_share = @JAP_PNR_ELECTION_THIRD_SEATS }
		JAP_pnr_election_add_shakai_taishuto_share = yes
		set_country_flag = JAP_pnr_election_rest_pool_2
		set_country_flag = JAP_pnr_election_rest_pool_8
		set_country_flag = JAP_pnr_election_rest_pool_9
		set_country_flag = JAP_pnr_election_rest_pool_10
		JAP_pnr_election_distribute_standard_rest_seats = yes
	}
	else_if = {
		limit = { JAP_pnr_election_party_6_leads = yes }
		set_variable = { JAP_pnr_election_winning_party = 6 }
		add_to_variable = { JAP_pnr_election_final_seats^6 = @JAP_PNR_ELECTION_WINNER_SEATS }
		add_to_variable = { JAP_pnr_election_final_seats^7 = @JAP_PNR_ELECTION_SECOND_SEATS }
		set_temp_variable = { JAP_pnr_election_shakai_taishuto_share = @JAP_PNR_ELECTION_THIRD_SEATS }
		JAP_pnr_election_add_shakai_taishuto_share = yes
		set_country_flag = JAP_pnr_election_rest_pool_2
		set_country_flag = JAP_pnr_election_rest_pool_8
		set_country_flag = JAP_pnr_election_rest_pool_9
		set_country_flag = JAP_pnr_election_rest_pool_10
		JAP_pnr_election_distribute_standard_rest_seats = yes
	}
	else = { # Shakai Taishuto (JAP_pnr_election_shakai_taishuto_leads)
		set_variable = { JAP_pnr_election_winning_party = 5 }
		add_to_variable = { JAP_pnr_election_final_seats^4 = @JAP_PNR_ELECTION_SHAKAI_TAISHUTO_WIN_SEATS_4 }
		add_to_variable = { JAP_pnr_election_final_seats^5 = @JAP_PNR_ELECTION_SHAKAI_TAISHUTO_WIN_SEATS_5 }
		JAP_pnr_election_assign_major_parties_by_score = yes
		set_country_flag = JAP_pnr_election_rest_pool_2
		set_country_flag = JAP_pnr_election_rest_pool_8
		set_country_flag = JAP_pnr_election_rest_pool_9
		set_country_flag = JAP_pnr_election_rest_pool_10
		JAP_pnr_election_distribute_standard_rest_seats = yes
	}

	if = {
		limit = { check_variable = { JAP_pnr_election_winning_party = 11 } }
		set_variable = { JAP_pnr_election_winning_party_seats = JAP_pnr_election_final_seats^7 }
		add_to_variable = { JAP_pnr_election_winning_party_seats = JAP_pnr_election_final_seats^6 }
	}
	else = {
		set_variable = { JAP_pnr_election_winning_party_seats = @JAP_PNR_ELECTION_WINNER_SEATS }
	}
}

# Writes the final result into the Imperial Diet panel. Call after JAP_pnr_election_calculate_winner.
JAP_pnr_election_commit_results = {
	JAP_pnr_election_ensure_projected_results = yes
