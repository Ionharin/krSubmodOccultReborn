JAP_pnr_election_initialize_projected_results = {
	set_country_flag = JAP_pnr_election_projected_initialized
	set_variable = { JAP_pnr_election_total_seats = 466 }

	set_variable = { JAP_pnr_election_party_seats^1 = 0 }
	set_variable = { JAP_pnr_election_party_seats^2 = 1 }
	set_variable = { JAP_pnr_election_party_seats^3 = 0 }
	set_variable = { JAP_pnr_election_party_seats^4 = 11 }
	set_variable = { JAP_pnr_election_party_seats^5 = 26 }
	set_variable = { JAP_pnr_election_party_seats^6 = 155 }
	set_variable = { JAP_pnr_election_party_seats^7 = 255 }
	set_variable = { JAP_pnr_election_party_seats^8 = 0 }
	set_variable = { JAP_pnr_election_party_seats^9 = 0 }
	set_variable = { JAP_pnr_election_party_seats^10 = 18 }

	set_variable = { JAP_pnr_election_party_seats_initial^2 = 1 }
	set_variable = { JAP_pnr_election_party_seats_initial^4 = 11 }
	set_variable = { JAP_pnr_election_party_seats_initial^5 = 26 }
	set_variable = { JAP_pnr_election_party_seats_initial^6 = 155 }
	set_variable = { JAP_pnr_election_party_seats_initial^7 = 255 }
	set_variable = { JAP_pnr_election_party_seats_initial^10 = 18 }

	JAP_pnr_election_refresh_projected_changes = yes
}

JAP_pnr_election_ensure_projected_results = {
	if = {
		limit = { NOT = { has_country_flag = JAP_pnr_election_projected_initialized } }
		JAP_pnr_election_initialize_projected_results = yes
	}
}

JAP_pnr_election_refresh_projected_donor = {
	set_variable = { JAP_pnr_election_donor_party = 2 }
	set_variable = { JAP_pnr_election_donor_party_seats = JAP_pnr_election_party_seats^2 }

	if = { limit = { check_variable = { JAP_pnr_election_party_seats^4 > JAP_pnr_election_donor_party_seats } } set_variable = { JAP_pnr_election_donor_party = 4 } set_variable = { JAP_pnr_election_donor_party_seats = JAP_pnr_election_party_seats^4 } }
	if = { limit = { check_variable = { JAP_pnr_election_party_seats^5 > JAP_pnr_election_donor_party_seats } } set_variable = { JAP_pnr_election_donor_party = 5 } set_variable = { JAP_pnr_election_donor_party_seats = JAP_pnr_election_party_seats^5 } }
	if = { limit = { check_variable = { JAP_pnr_election_party_seats^6 > JAP_pnr_election_donor_party_seats } } set_variable = { JAP_pnr_election_donor_party = 6 } set_variable = { JAP_pnr_election_donor_party_seats = JAP_pnr_election_party_seats^6 } }
	if = { limit = { check_variable = { JAP_pnr_election_party_seats^7 > JAP_pnr_election_donor_party_seats } } set_variable = { JAP_pnr_election_donor_party = 7 } set_variable = { JAP_pnr_election_donor_party_seats = JAP_pnr_election_party_seats^7 } }
}

JAP_pnr_election_refresh_projected_changes = {
	set_variable = { JAP_pnr_election_party_change^2 = JAP_pnr_election_party_seats^2 }
	subtract_from_variable = { JAP_pnr_election_party_change^2 = JAP_pnr_election_party_seats_initial^2 }
	set_variable = { JAP_pnr_election_party_change^4 = JAP_pnr_election_party_seats^4 }
	subtract_from_variable = { JAP_pnr_election_party_change^4 = JAP_pnr_election_party_seats_initial^4 }
	set_variable = { JAP_pnr_election_party_change^5 = JAP_pnr_election_party_seats^5 }
	subtract_from_variable = { JAP_pnr_election_party_change^5 = JAP_pnr_election_party_seats_initial^5 }
	set_variable = { JAP_pnr_election_party_change^6 = JAP_pnr_election_party_seats^6 }
	subtract_from_variable = { JAP_pnr_election_party_change^6 = JAP_pnr_election_party_seats_initial^6 }
	set_variable = { JAP_pnr_election_party_change^7 = JAP_pnr_election_party_seats^7 }
	subtract_from_variable = { JAP_pnr_election_party_change^7 = JAP_pnr_election_party_seats_initial^7 }
	set_variable = { JAP_pnr_election_party_change^10 = JAP_pnr_election_party_seats^10 }
	subtract_from_variable = { JAP_pnr_election_party_change^10 = JAP_pnr_election_party_seats_initial^10 }
}

JAP_pnr_election_party_10_gain_10_seats = {
	JAP_pnr_election_ensure_projected_results = yes
	JAP_pnr_election_refresh_projected_donor = yes

	set_variable = { JAP_pnr_election_seats_to_move = 10 }
	if = {
		limit = { check_variable = { JAP_pnr_election_donor_party_seats < JAP_pnr_election_seats_to_move } }
		set_variable = { JAP_pnr_election_seats_to_move = JAP_pnr_election_donor_party_seats }
	}

	subtract_from_variable = { JAP_pnr_election_party_seats^JAP_pnr_election_donor_party = JAP_pnr_election_seats_to_move }
	add_to_variable = { JAP_pnr_election_party_seats^10 = JAP_pnr_election_seats_to_move }
	JAP_pnr_election_refresh_projected_changes = yes

	custom_effect_tooltip = JAP_pnr_election_party_10_gain_10_seats_tt
}

	set_variable = { JAP_congress_faction_seats_list^9 = 0 }
