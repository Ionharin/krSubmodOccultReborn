JAP_pnr_fleet_plan_ensure_modifier = {
	if = {
		limit = {
			NOT = {
				has_dynamic_modifier = {
					modifier = JAP_pnr_fleet_plan_modifier
				}
			}
		}
		set_variable = { JAP_pnr_fleet_plan_experience_gain_navy_factor = 0.15 }
		set_variable = { JAP_pnr_fleet_plan_navy_org_factor = 0.05 }
		add_dynamic_modifier = {
			modifier = JAP_pnr_fleet_plan_modifier
		}
	}
}
