    }

    JAP_pnr_add_to_yen_economic_sphere = {
        if = {
            limit = {
                NOT = { tag = JAP }
                JAP = { is_economic_sphere_leader = yes }
            }
            if = {
                limit = {
                    is_in_economic_sphere = yes
                    JAP = { is_leader_of_economic_sphere_of_PREV = no }
                }
                leave_economic_sphere = yes
            }
            if = {
                limit = { JAP = { is_leader_of_economic_sphere_of_PREV = no } }
                set_temp_variable = { sphere_target = THIS }
                JAP = { add_to_economic_sphere = yes }
            }
            if = {
                limit = { NOT = { has_dynamic_modifier = { modifier = JAP_pnr_yen_economic_sphere_member } } }
                add_dynamic_modifier = { modifier = JAP_pnr_yen_economic_sphere_member }
            }
            if = {
                limit = { NOT = { is_in_tech_sharing_group = japan_research } }
                add_to_tech_sharing_group = japan_research
            }
            JAP = { JAP_pnr_recalculate_yen_economic_sphere_economic_integration = yes }
        }
    }

    JAP_pnr_remove_from_yen_economic_sphere = {
        if = {
            limit = { JAP = { is_leader_of_economic_sphere_of_PREV = yes } }
            leave_economic_sphere = yes
            if = {
                limit = { has_dynamic_modifier = { modifier = JAP_pnr_yen_economic_sphere_member } }
                remove_dynamic_modifier = { modifier = JAP_pnr_yen_economic_sphere_member }
            }
            if = {
                limit = { is_in_tech_sharing_group = japan_research }
                remove_from_tech_sharing_group = japan_research
            }
            JAP = { JAP_pnr_recalculate_yen_economic_sphere_economic_integration = yes }
        }
    }

    JAP_pnr_refresh_yen_economic_sphere_members = {
        if = {
            limit = { is_economic_sphere_leader = yes }
            if = {
                limit = { NOT = { is_in_tech_sharing_group = japan_research } }
                add_to_tech_sharing_group = japan_research
            }
            JAP_pnr_yen_economic_sphere_display = yes
            for_each_scope_loop = {
                array = JAP.economic_sphere:members
                if = {
                    limit = { NOT = { is_in_tech_sharing_group = japan_research } }
                    add_to_tech_sharing_group = japan_research
                }
                if = {
                    limit = {
                        NOT = { tag = JAP }
                        NOT = { is_subject_of = JAP }
                        NOT = { is_in_faction_with = JAP }
                    }
                    JAP_pnr_remove_from_yen_economic_sphere = yes
                }
            }
            every_subject_country = {
                limit = {
                    exists = yes
                    NOT = { tag = JAP }
                }
                JAP_pnr_add_to_yen_economic_sphere = yes
            }
            every_other_country = {
                limit = {
                    exists = yes
                    is_in_faction_with = JAP
                    NOT = { tag = JAP }
                }
                JAP_pnr_add_to_yen_economic_sphere = yes
            }
        }
    }

    JAP_pnr_recalculate_yen_economic_sphere_economic_integration = {
        if = {
            limit = { is_economic_sphere_leader = yes }
            hidden_effect = {
                set_temp_variable = { JAP_yen_economic_sphere_factories = 0 }
                for_each_scope_loop = {
                    array = JAP.economic_sphere:members
                    add_to_temp_variable = { JAP_yen_economic_sphere_factories = THIS.num_of_civilian_factories }
                }
