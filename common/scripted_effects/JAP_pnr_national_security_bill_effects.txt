JAP_pnr_mio_check_industrial_complex = {
	set_temp_variable = {
		JAP_pnr_mio_check_industrial_complex_var = FROM.num_of_civilian_factories
		tooltip = JAP_mio_check_industrial_complex_tt
	}
	if = {
		limit = { check_variable = { JAP_pnr_mio_check_industrial_complex_var < 100 } }
		multiply_temp_variable = { JAP_pnr_mio_check_industrial_complex_var = 0.0010 }
		add_mio_research_bonus = JAP_pnr_mio_check_industrial_complex_var
	}
	else = {
		add_mio_research_bonus = 0.10
	}
}
