	add_to_variable = { JAP_pnr_landreform_current_time_2 = JAP_pnr_landreform_time_2 }
	set_variable = { JAP_pnr_landreform_current_time_3 = JAP_pnr_peasant_angry_var }

	add_to_variable = { JAP_pnr_landreform_current_time_3 = JAP_pnr_landreform_time_3 }

JAP_pnr_landreform_check_completion = {
	clamp_variable = { var = JAP_pnr_landreform_progress_var max = 100 min = 0 }
	clamp_variable = { var = JAP_pnr_landreform_result_var max = 100 min = 0 }
	clamp_variable = { var = JAP_pnr_landowner_angry_var max = 10 min = 0 }
	clamp_variable = { var = JAP_pnr_peasant_angry_var max = 10 min = 0 }
	if = {
		limit = { check_variable = { JAP_pnr_landreform_progress_var > 99 } }
		set_country_flag = pnr_landreform_end_flag
	}
	JAP_pnr_update_landreform_costs = yes
}

JAP_pnr_apply_landreform_landowner_backlash = {
	if = {
		limit = { check_variable = { JAP_pnr_landowner_angry_var > 0 } }
		set_temp_variable = { JAP_pnr_landreform_landowner_stability_penalty = JAP_pnr_landowner_angry_var }
		multiply_temp_variable = { JAP_pnr_landreform_landowner_stability_penalty = -0.005 }
		add_stability = var:JAP_pnr_landreform_landowner_stability_penalty
		add_to_variable = {
			var = JAP_pnr_landreform_progress_var
			value = JAP_pnr_landowner_angry_var
			tooltip = JAP_pnr_landreform_progress_var_raised
		}
	}
}
