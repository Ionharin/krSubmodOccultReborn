JAP_congress_rebuild_seats = {
	JAP_congress_set_seats_position = yes
	clear_array = JAP_congress_seats_array
	for_each_loop = {
		array = JAP_congress_faction_list
		value = faction_num
		set_temp_variable = { temp_seats = JAP_congress_faction_seats_list^faction_num }
		for_loop_effect = {
			start = 0
			end = temp_seats
			add_to_array = { JAP_congress_seats_array = faction_num }
		}
	}
	JAP_congress_update_gui = yes
}

JAP_congress_move_seats = {
	subtract_from_variable = {
		JAP_congress_faction_seats_list^faction_id = seats_amount
	}
	add_to_variable = {
		JAP_congress_faction_seats_list^move_to = seats_amount
	}
	JAP_congress_rebuild_seats = yes
}
