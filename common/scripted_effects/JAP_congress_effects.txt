JAP_congress_set_initial_authority = {
	JAP_congress_refresh_ruling_party = yes
	set_variable = { JAP_congress_authority_initial = JAP_congress_majority_party_seats }
	if = {
		limit = { check_variable = { JAP_congress_authority_initial > 234 } }
		set_variable = { JAP_congress_authority_initial = 234 }
	}
	set_variable = { JAP_congress_authority = JAP_congress_authority_initial }
}

JAP_congress_add_authority = {
	set_temp_variable = { JAP_congress_authority_value_tooltip = 234 }
	subtract_from_temp_variable = { JAP_congress_authority_value_tooltip = JAP_congress_authority }
	if = {
		limit = { check_variable = { JAP_congress_authority_value_tooltip > JAP_congress_authority_value } }
		set_temp_variable = { JAP_congress_authority_value_tooltip = JAP_congress_authority_value }
	}
	add_to_variable = { JAP_congress_authority = JAP_congress_authority_value_tooltip }
	if = {
		limit = { check_variable = { JAP_congress_authority > 234 } }
		set_variable = { JAP_congress_authority = 234 }
	}
	set_temp_variable = { JAP_congress_authority_cost = JAP_congress_authority_value_tooltip }
	multiply_temp_variable = { JAP_congress_authority_cost = 3 }
	multiply_temp_variable = { JAP_congress_authority_cost = -1 }
	set_temp_variable = { JAP_congress_authority_cost_tooltip = JAP_congress_authority_value_tooltip }
	multiply_temp_variable = { JAP_congress_authority_cost_tooltip = 3 }
	hidden_effect = { add_political_power = var:JAP_congress_authority_cost }
	JAP_congress_update_gui = yes
	custom_effect_tooltip = JAP_congress_authority_add_tooltip
}

JAP_congress_subtract_authority = {
	set_temp_variable = { JAP_congress_authority_value_tooltip = JAP_congress_authority_value }
	subtract_from_variable = { JAP_congress_authority = JAP_congress_authority_value }

		limit = { check_variable = { JAP_congress_authority < JAP_congress_authority_initial } }
		set_variable = { JAP_congress_authority = JAP_congress_authority_initial }

	set_temp_variable = { JAP_congress_authority_cost_tooltip = JAP_congress_authority_value_tooltip }
	multiply_temp_variable = { JAP_congress_authority_cost_tooltip = 3 }
	hidden_effect = {
		add_political_power = var:JAP_congress_authority_cost_tooltip
	}
	JAP_congress_update_gui = yes
	custom_effect_tooltip = JAP_congress_authority_subtract_tooltip
}
