JAP_pnr_initialize_interservice_budget = { # Set Military budget and other variable on startup
	if = {
		limit = {
			tag = JAP
			NOT = { has_country_flag = JAP_pnr_interservice_budget_begun }
		}
		set_country_flag = JAP_pnr_interservice_budget_begun
		set_variable = { JAP_pnr_army_budget_points = 2 }
		set_variable = { JAP_pnr_navy_budget_points = 1 }
		set_variable = { JAP_pnr_army_budget_yearly_gain = 3 }
		set_variable = { JAP_pnr_navy_budget_yearly_gain = 2 }
		set_variable = { JAP_pnr_lost_decade_consumer_goods_factor = 0.09 }
		set_variable = { JAP_pnr_lost_decade_stability_factor = -0.16 }
		set_variable = { JAP_pnr_lost_decade_production_speed_buildings_factor = -0.15 }
		set_variable = { JAP_pnr_lost_decade_production_factory_max_efficiency_factor = -0.14 }
		set_variable = { JAP_pnr_lost_decade_industrial_capacity_factory = -0.17 }
		set_variable = { JAP_pnr_lost_decade_industrial_capacity_dockyard = -0.17 }
		set_variable = { JAP_pnr_lost_decade_disabled_ideas = 1 }
		set_variable = { JAP_pnr_fleet_plan_experience_gain_navy_factor = 0.15 }
		set_variable = { JAP_pnr_fleet_plan_navy_org_factor = 0.05 }
		set_variable = { JAP_land_reform_cost_1 = 50 }
		set_variable = { JAP_land_reform_cost_2 = 75 }
		set_variable = { JAP_land_reform_cost_3 = 100 }
		set_variable = { JAP_land_reform_cost_4 = 150 }
		set_variable = { JAP_land_reform_pp_1 = 20 }
		set_variable = { JAP_land_reform_pp_2 = 35 }
		set_variable = { JAP_land_reform_pp_3 = 50 }
		set_variable = { JAP_pnr_zaibatsu_government_path_var = 0 }
		activate_mission = JAP_pnr_next_year_interservice_budget_mission
	}
}

JAP_pnr_add_annual_interservice_budget = {
	add_to_variable = {
		var = JAP_pnr_army_budget_points
		value = JAP_pnr_army_budget_yearly_gain
	}
	add_to_variable = {
		var = JAP_pnr_navy_budget_points
		value = JAP_pnr_navy_budget_yearly_gain
	}
}

JAP_pnr_subtract_army_budget_point = {
	subtract_from_variable = {
		JAP_pnr_army_budget_points = 1
	}
}

JAP_pnr_subtract_2_army_budget_point = {
	subtract_from_variable = {
		JAP_pnr_army_budget_points = 2
	}
}

JAP_pnr_subtract_navy_budget_point = {
	subtract_from_variable = {
		JAP_pnr_navy_budget_points = 1
	}
}

JAP_pnr_subtract_2_navy_budget_point = {
	subtract_from_variable = {
		JAP_pnr_navy_budget_points = 2
	}
}
