	clamp_variable = { var = JAP_pnr_landreform_progress_var max = 100 min = 0 }
	clamp_variable = { var = JAP_pnr_landreform_result_var max = 100 min = 0 }
	clamp_variable = { var = JAP_pnr_landowner_angry_var max = 10 min = 0 }
	clamp_variable = { var = JAP_pnr_peasant_angry_var max = 10 min = 0 }
	if = {
		limit = { check_variable = { JAP_pnr_landreform_progress_var > 99 } }
		set_country_flag = pnr_landreform_end_flag
	}
	JAP_pnr_update_landreform_costs = yes
}

			value = JAP_pnr_landowner_angry_var

		}

		set_variable = { JAP_pnr_red_conspiracy_strength = 15 }

}

		limit = { check_variable = { JAP_pnr_mio_check_industrial_complex_var < 100 } }
