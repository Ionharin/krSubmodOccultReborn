JAP_pnr_initialize_new_party_preparation = {
	set_variable = { JAP_pnr_kn_new_party_supporters = 11 }
	set_variable = { JAP_pnr_kn_reform_bureaucrats = 8 }
	set_variable = { JAP_pnr_kn_military_associates = 17 }
	set_variable = { JAP_pnr_kn_party_politicians = 57 }
	set_variable = { JAP_pnr_kn_court_forces = 7 }
	JAP_pnr_update_new_party_preparation_gui = yes
}

JAP_pnr_initialize_new_party_preparation_stances = {
	set_variable = { JAP_pnr_kn_new_party_supporters_stance = 1 }
	set_variable = { JAP_pnr_kn_reform_bureaucrats_stance = 0 }
	set_variable = { JAP_pnr_kn_military_associates_stance = 0 }
	set_variable = { JAP_pnr_kn_party_politicians_stance = -1 }
	set_variable = { JAP_pnr_kn_court_forces_stance = 1 }
}

JAP_pnr_update_new_party_preparation_gui = {
	clamp_variable = { var = JAP_pnr_kn_new_party_supporters max = 100 min = 0 }
	clamp_variable = { var = JAP_pnr_kn_reform_bureaucrats max = 100 min = 0 }
	clamp_variable = { var = JAP_pnr_kn_military_associates max = 100 min = 0 }
	clamp_variable = { var = JAP_pnr_kn_party_politicians max = 100 min = 0 }
	clamp_variable = { var = JAP_pnr_kn_court_forces max = 100 min = 0 }

	set_variable = { JAP_pnr_kn_gui_new_party_supporters = JAP_pnr_kn_new_party_supporters }
	set_variable = { JAP_pnr_kn_gui_reform_bureaucrats = { value = JAP_pnr_kn_gui_new_party_supporters add = JAP_pnr_kn_reform_bureaucrats } }
	set_variable = { JAP_pnr_kn_gui_military_associates = { value = JAP_pnr_kn_gui_reform_bureaucrats add = JAP_pnr_kn_military_associates } }
	set_variable = { JAP_pnr_kn_gui_party_politicians = { value = JAP_pnr_kn_gui_military_associates add = JAP_pnr_kn_party_politicians } }
	set_variable = { JAP_pnr_kn_gui_court_forces = { value = JAP_pnr_kn_gui_party_politicians add = JAP_pnr_kn_court_forces } }

	clamp_variable = { var = JAP_pnr_kn_gui_new_party_supporters max = 100 min = 0 }
	clamp_variable = { var = JAP_pnr_kn_gui_reform_bureaucrats max = 100 min = 0 }
	clamp_variable = { var = JAP_pnr_kn_gui_military_associates max = 100 min = 0 }
	clamp_variable = { var = JAP_pnr_kn_gui_party_politicians max = 100 min = 0 }
	clamp_variable = { var = JAP_pnr_kn_gui_court_forces max = 100 min = 0 }
}
