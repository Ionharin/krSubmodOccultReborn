defined_text = { 
	name = GetSICName
	text = {
		trigger = {
			any_character = {
				#is_second_in_command = yes
				has_character_flag = head_of_country_flag
				set_temp_variable = { current_sic_character = THIS }
			}
		}
		localization_key = "[?current_sic_character.GetName]"
	}

}

defined_text = { 
	name = GetHOSTittle

	text = {
		trigger = {
			any_character = {
				is_country_leader = yes
				NOT = {
					check_variable = {
						SICtittle = 0
					}
				}
				set_temp_variable = {
					temp_tittle_sic = var:SICtittle
				}
				PREV = {
					set_temp_variable = {
						gui_t_hos = var:temp_tittle_sic
					}
				}
			}
		}
		localization_key = "KRM_sic_tittle_[?gui_t_hos]"
	}

	text = {
		trigger = {}
		localization_key = "KRM_sic_tittle_0"
	}

}
