JAP_congress_has_political_power_for_authority = {
	set_temp_variable = { JAP_congress_authority_value_actual = 234 }
	subtract_from_temp_variable = { JAP_congress_authority_value_actual = JAP_congress_authority }
	if = {
		limit = { check_variable = { JAP_congress_authority_value_actual > JAP_congress_authority_value } }
		set_temp_variable = { JAP_congress_authority_value_actual = JAP_congress_authority_value }
	}
	multiply_temp_variable = { JAP_congress_authority_value_actual = 3 }
	custom_override_tooltip = {
		tooltip = JAP_congress_authority_at_maximum_tt
		check_variable = { JAP_congress_authority < 234 }
	}
	custom_override_tooltip = {
		tooltip = JAP_congress_authority_not_enough_pp_tt
		NOT = { check_variable = { political_power < JAP_congress_authority_value_actual } }
	}
}

JAP_congress_can_recover_authority = {
	set_temp_variable = { JAP_congress_authority_recoverable = JAP_congress_authority }
	subtract_from_temp_variable = { JAP_congress_authority_recoverable = JAP_congress_authority_initial }
	custom_override_tooltip = {
		tooltip = JAP_congress_authority_at_minimum_tt
		NOT = { check_variable = { JAP_congress_authority_recoverable < JAP_congress_authority_value } }
	}
}
