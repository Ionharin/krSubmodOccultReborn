JAP_pnr_has_army_budget_point_trigger = { # subtract 1 army budget point effect
	custom_override_tooltip = {
		tooltip = JAP_pnr_army_budget_point_required_tt
		check_variable = {
			var = JAP_pnr_army_budget_points
			value = 1
			compare = greater_than_or_equals
		}
	}
}
